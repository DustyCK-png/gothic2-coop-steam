cmake_minimum_required(VERSION 3.24)
project(Gothic2Coop LANGUAGES CXX)
set(SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/GothicCoop")
add_library(GothicCoop_G2A SHARED "${SOURCE}/Interface.cpp" "${SOURCE}/DllMain.cpp")
file(GLOB G2GLOBALS "${SOURCE}/ZenGin/Gothic_II_Addon/*_G2A.cpp")
target_sources(GothicCoop_G2A PRIVATE ${G2GLOBALS})
target_compile_features(GothicCoop_G2A PRIVATE cxx_std_17)
target_compile_definitions(GothicCoop_G2A PRIVATE __G2A WIN32 _WINDOWS _USRDLL GOTHICPLUGIN_EXPORTS _WINSOCK_DEPRECATED_NO_WARNINGS)
target_include_directories(GothicCoop_G2A PRIVATE "${SOURCE}" "${SOURCE}/ZenGin/Gothic_UserAPI" "${SOURCE}/SDK/Union/v1.0l/Include" "${SOURCE}/SDK/DirectX7/include" "${SOURCE}/dependencies/enet-1.3.17/include" "${SOURCE}/dependencies/nlohmann-json/include/single_include")
target_link_directories(GothicCoop_G2A PRIVATE "${SOURCE}/SDK/Union/v1.0l/Lib" "${SOURCE}/SDK/DirectX7/Lib" "${SOURCE}/dependencies/enet-1.3.17")
target_link_libraries(GothicCoop_G2A PRIVATE enet ws2_32 winmm dbghelp gdiplus)
target_link_options(GothicCoop_G2A PRIVATE /SAFESEH:NO)
