cmake_minimum_required(VERSION 3.24)
project(G2Steam C CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Steam message ABI requires Win32")
endif()
add_executable(G2Steam g2steam.c config.cpp)
target_compile_features(G2Steam PRIVATE cxx_std_17)
target_include_directories(G2Steam PRIVATE ../vendor/GothicCoop/dependencies/nlohmann-json/include/single_include)
add_library(mock_steam_api SHARED mock_steam_api.c)
foreach(target G2Steam mock_steam_api)
 target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_link_libraries(${target} PRIVATE ws2_32)
endforeach()
